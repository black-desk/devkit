cmake_minimum_required(VERSION 3.20)
project(seed_runtime_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_library(throwing SHARED plugin.cpp)
add_executable(probe probe.cpp)
target_link_libraries(probe PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME runtime COMMAND probe "${DEVKIT_PREFIX}" "$<TARGET_FILE:throwing>")
set_tests_properties(runtime PROPERTIES TIMEOUT 30)
