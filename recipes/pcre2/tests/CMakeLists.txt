cmake_minimum_required(VERSION 3.15)
project(devkit_pcre2_consumer C)
find_package(PCRE2 CONFIG REQUIRED COMPONENTS 8BIT 16BIT 32BIT POSIX)
foreach(width 8 16 32)
  add_executable(probe-${width} probe.c)
  target_compile_definitions(probe-${width} PRIVATE PCRE2_CODE_UNIT_WIDTH=${width})
  target_link_libraries(probe-${width} PRIVATE PCRE2::${width}BIT)
endforeach()
add_executable(probe-posix posix.c)
target_link_libraries(probe-posix PRIVATE PCRE2::POSIX)
