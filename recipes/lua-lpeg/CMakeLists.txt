cmake_minimum_required(VERSION 3.16)
project(lpeg C)
include(GNUInstallDirs)
file(GLOB sources "${CMAKE_CURRENT_SOURCE_DIR}/*.c")
add_library(lpeg SHARED ${sources})
target_include_directories(lpeg PRIVATE "${LUAJIT_INCLUDE_DIR}")
target_link_libraries(lpeg PRIVATE "${LUAJIT_LIBRARY}")
set_target_properties(lpeg PROPERTIES VERSION 1.1.0 SOVERSION 1)
install(TARGETS lpeg LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}")
