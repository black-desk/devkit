cmake_minimum_required(VERSION 3.16)
project(parser C)
file(GLOB markdown_sources "tree-sitter-markdown/src/*.c")
add_library(markdown MODULE ${markdown_sources})
target_include_directories(markdown PRIVATE "tree-sitter-markdown/src")
set_target_properties(markdown PROPERTIES PREFIX "")
install(TARGETS markdown LIBRARY DESTINATION lib/devkit/lib/nvim/parser)
file(GLOB markdown_inline_sources "tree-sitter-markdown-inline/src/*.c")
add_library(markdown_inline MODULE ${markdown_inline_sources})
target_include_directories(markdown_inline PRIVATE "tree-sitter-markdown-inline/src")
set_target_properties(markdown_inline PROPERTIES PREFIX "")
install(TARGETS markdown_inline LIBRARY DESTINATION lib/devkit/lib/nvim/parser)
