cmake_minimum_required(VERSION 3.20)
project(devkit_discovery NONE)
find_package(DevkitDiscovery CONFIG QUIET)
find_library(DEVKIT_DISCOVERY_LIBRARY NAMES devkit-discovery)
find_path(DEVKIT_DISCOVERY_INCLUDE NAMES devkit-discovery.h)
find_program(DEVKIT_DISCOVERY_TOOL NAMES devkit-discovery-tool REQUIRED)
foreach(result DevkitDiscovery_FOUND DEVKIT_DISCOVERY_LIBRARY DEVKIT_DISCOVERY_INCLUDE)
  if(EXPECT_CHANNEL AND NOT ${result})
    message(FATAL_ERROR "Explicit channel lookup failed: ${result}")
  elseif(NOT EXPECT_CHANNEL AND ${result})
    message(FATAL_ERROR "Default lookup found channel development files: ${result}")
  endif()
endforeach()
if(EXPECT_PREFIX)
  foreach(result DevkitDiscovery_DIR DEVKIT_DISCOVERY_LIBRARY DEVKIT_DISCOVERY_INCLUDE)
    string(FIND "${${result}}" "${EXPECT_PREFIX}/" position)
    if(NOT position EQUAL 0)
      message(FATAL_ERROR "Unexpected lookup location: ${result}=${${result}}")
    endif()
  endforeach()
endif()
